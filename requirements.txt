# Telegram client — MTProto, not the Bot API.
# This is the whole reason big files work: the cloud Bot API caps uploads at
# 50 MB, MTProto allows 2 GB (4 GB on a Premium account).
pyrogram==2.0.106
tgcrypto==1.2.5

# HTTP with a real browser TLS fingerprint. Plain `requests` gets 403'd by the
# CDNs these links live behind.
curl-cffi==0.7.4

# QR rendering for the payment screen. The Node gateway hands us a UPI URI and
# the QR's own matrix; Telegram cannot send SVG as a photo, so `payments.py` draws
# the PNG itself from that matrix. No `qrcode` encoder here on purpose — a second
# encoder in this process could draw a perfectly valid code for the wrong amount.
pillow==10.4.0

# VPS status panel in the admin menu.
psutil==6.0.0

# Password-protected ZIPs (AES). The stdlib zipfile only handles legacy ZipCrypto.
pyzipper==0.3.6

# RAR. `rarfile` is a wrapper, not a decoder — it needs a binary, and the one to
# install is `unar` (apt install unar), NOT `unrar`: unrar is non-free and is not
# in Ubuntu's default repositories, while unar reads RAR5 too.
rarfile==4.2

# 7z, pure Python, so nothing outside pip has to be on the box.
py7zr==0.22.0

# Postgres, for when DATABASE_URL is set — which is how credits outlive the VPS
# they were bought on. Unused and unimported on a default install: `bot/db.py`
# imports it inside the branch that needs it, so a SQLite box never loads it.
# The [binary] extra ships libpq, so there is no apt package to remember.
psycopg[binary]==3.2.1
